df_mod = df_clean.copy()
salary_norm = (df_mod['Salary_INR'] - df_mod['Salary_INR'].min()) / (df_mod['Salary_INR'].max() - df_mod['Salary_INR'].min())
perf_norm = (df_mod['Performance_Rating'] - df_mod['Performance_Rating'].min()) / (df_mod['Performance_Rating'].max() - df_mod['Performance_Rating'].min())
np.random.seed(42)
risk_score = (0.6 * (1 - salary_norm)) + (0.2 * (1 - perf_norm)) + (0.2 * np.random.rand(len(df_mod)))
df_mod['Attrition_Flag'] = np.where(risk_score > 0.6, 1, 0)

# Danh sách các model cần chạy
models = {
    "Logistic Regression": LogisticRegression(max_iter=1000),
    "Random Forest": RandomForestClassifier(n_estimators=100, random_state=42),
    "Gradient Boosting": GradientBoostingClassifier(random_state=42)
}

print("=== BẮT ĐẦU CHẠY VÀ PHÂN TÍCH TỪNG MODEL ===")

if hasattr(X_train_res_new, 'columns'):
    feature_names = X_train_res_new.columns
else:
    feature_names = X_new.columns

for name, model in models.items():
    print(f"\n{'='*80}")
    print(f"⏩ MODEL: {name.upper()}")
    
    # 1. Huấn luyện
    model.fit(X_train_res_new, y_train_res_new)
    y_pred = model.predict(X_test_new)
    
    # 2. Đánh giá
    acc = accuracy_score(y_test_new, y_pred)
    roc = roc_auc_score(y_test_new, y_pred)
    print(f"► Độ chính xác (Accuracy): {acc:.2%}")
    print(f"► ROC AUC Score: {roc:.4f}")

    # 3. Vẽ biểu đồ (1 Hàng, 2 Cột)
    fig, axes = plt.subplots(1, 2, figsize=(20, 7))
    
    # --- BIỂU ĐỒ TRÁI: CONFUSION MATRIX ---
    cm = confusion_matrix(y_test_new, y_pred)
    group_names = ['True Neg\n(Đúng: Ở lại)', 'False Pos\n(Sai: Báo giả)', 
                   'False Neg\n(Sai: Bỏ sót)', 'True Pos\n(Đúng: Nghỉ việc)']
    group_counts = ["{0:0,d}".format(value) for value in cm.flatten()]
    group_percentages = ["{0:.2%}".format(value) for value in cm.flatten()/np.sum(cm)]
    labels = [f"{v1}\n{v2}\n({v3})" for v1, v2, v3 in zip(group_names, group_counts, group_percentages)]
    labels = np.asarray(labels).reshape(2,2)
    
    sns.heatmap(cm, annot=labels, fmt='', cmap='Blues', cbar=False, 
                annot_kws={"size": 12, "weight": "bold"}, ax=axes[0])
    axes[0].set_title(f'CONFUSION MATRIX - {name}', fontsize=14, fontweight='bold', color='darkblue')
    axes[0].set_xlabel('Dự đoán của máy')
    axes[0].set_ylabel('Thực tế')
    axes[0].set_xticklabels(['Ở lại', 'Nghỉ việc'])
    axes[0].set_yticklabels(['Ở lại', 'Nghỉ việc'], rotation=0)
    # --- BIỂU ĐỒ PHẢI: MỨC ĐỘ QUAN TRỌNG ---
    if name == "Logistic Regression":
        importances = model.coef_[0]
        imp_df = pd.DataFrame({'Feature': feature_names, 'Value': importances})
        imp_df['Abs_Val'] = imp_df['Value'].abs()
        imp_df = imp_df.sort_values(by='Abs_Val', ascending=False).head(10)
        sns.barplot(x='Value', y='Feature', data=imp_df, palette='coolwarm', ax=axes[1])
        axes[1].set_title(f'YẾU TỐ TÁC ĐỘNG (HỆ SỐ) - {name}', fontsize=14, fontweight='bold')
        axes[1].axvline(x=0, color='black', linestyle='--')
    else:
        importances = model.feature_importances_
        imp_df = pd.DataFrame({'Feature': feature_names, 'Value': importances})
        imp_df = imp_df.sort_values(by='Value', ascending=False).head(10)
        sns.barplot(x='Value', y='Feature', data=imp_df, palette='viridis', ax=axes[1])
        axes[1].set_title(f'TOP 10 YẾU TỐ QUAN TRỌNG - {name}', fontsize=14, fontweight='bold')

    plt.tight_layout()
    plt.show()
